    76→    76→function drawStroke(ctx: CanvasRenderingContext2D, s: Stroke) {
    77→    77→  if (s.points.length === 0) return;
    78→    78→  ctx.save();
    79→    79→  if (s.kind === "highlighter") {
    80→    80→    ctx.globalAlpha = 0.35;
    81→    81→    ctx.globalCompositeOperation = "multiply";
    82→    82→  }
    83→    83→  ctx.strokeStyle = s.color;
    84→    84→  ctx.lineWidth = s.width;
    85→    85→  ctx.lineCap = "round";
    86→    86→  ctx.lineJoin = "round";
    87→    87→  ctx.beginPath();
    88→    88→  ctx.moveTo(s.points[0].x, s.points[0].y);
    89→    89→  for (let i = 1; i < s.points.length; i++) {
    90→    90→    ctx.lineTo(s.points[i].x, s.points[i].y);
    91→    91→  }
    92→    92→  ctx.stroke();
    93→    93→  ctx.restore();
    94→    94→}

   400→   400→      }

   408→   408→        x,
   409→   409→        y,
   410→   410→        w: 0,
   411→   411→        h: 0,
   412→   412→        color: "var(--brand)",
   413→   413→        fill: false,
   414→   414→      });

   421→   421→    const { x, y } = getCanvasPos(e);
   422→   422→    if (tool2 === "draw" || tool2 === "highlighter") {
   423→   423→      setDraft((d) => {
   424→   424→        if (!d || !(d.kind === "draw" || d.kind === "highlighter")) return d;
   425→   425→        const s = d as Stroke;
   426→   426→        return { ...s, points: [...s.points, { x, y }] };
   427→   427→      });
   428→   428→    } else if (tool2 === "rect" || tool2 === "circle" || tool2 === "crop") {
   429→   429→      const start = draftStartRef.current;
   430→   430→      setDraft((d) => {
   431→   431→        if (!d) return d;
   432→   432→        const sh = d as Shape;
   433→   433→        return {
   434→   434→          ...sh,
   435→   435→          x: Math.min(start.x, x),
   436→   436→          y: Math.min(start.y, y),
   437→   437→          w: Math.abs(x - start.x),
   438→   438→          h: Math.abs(y - start.y),
   439→   439→        };
   440→   440→      });
   441→   441→    }
   442→   442→  };
   443→   443→
   444→   444→  const onPointerUp = (e: React.PointerEvent) => {
   445→   445→    drawingRef.current = false;
   446→   446→    (e.target as HTMLElement).releasePointerCapture?.(e.pointerId);
   447→   447→    if (!draftStartRef.current) return;
   448→   448→    if (tool2 === "draw" || tool2 === "highlighter") {
   449→   449→      setDraft((d) => {
   450→   450→        if (d && (d.kind === "draw" || d.kind === "highlighter")) {
   451→   451→          setStrokes((prev) => [...prev, d as Stroke]);
   452→   452→        }
   453→   453→        return null;
   454→   454→      });
   455→   455→    } else if (tool2 === "rect" || tool2 === "circle") {
   456→   456→      setDraft((d) => {
   457→   457→        if (d && (d.kind === "rect" || d.kind === "circle")) {
   458→   458→          const sh = d as Shape;
   459→   459→          if (sh.w > 4 && sh.h > 4) {
   460→   460→            setShapes((prev) => [...prev, sh]);
   461→   461→          }
   462→   462→        }
   463→   463→        return null;
   464→   464→      });
   465→   465→    }
   466→   466→    draftStartRef.current = null;
   467→   467→  };
   468→   468→
   469→   469→  const commitTextDraft = () => {
