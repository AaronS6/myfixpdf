   336→/** Intersection Observer hook — returns true once the element enters the viewport. */
   337→function useInView<T extends HTMLElement>(): [React.RefObject<T | null>, boolean] {
   338→  const ref = useRef<T>(null);
   339→  const [inView, setInView] = useState(false);
   340→  useEffect(() => {
   341→    const el = ref.current;
   342→    if (!el) return;
   343→    const ro = new IntersectionObserver(
   344→      ([entry]) => { if (entry.isIntersecting) { setInView(true); ro.disconnect(); } },
   345→      { threshold: 0.15 },
   346→    );
   347→    ro.observe(el);
   348→    return () => ro.disconnect();
   349→  }, []);
   350→  return [ref, inView];
   351→}
