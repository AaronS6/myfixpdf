    59→    59→
    60→    60→export function EditPdf() {
    61→    61→  const { sourceFiles, setResult, setView, startProgress, updateProgress, stopProgress, pushHistory, popHistory } = useDocumentSession();
    62→    62→  const [liveBlob, setLiveBlob] = useState<Blob | null>(null);
    63→    63→  const [pageNum, setPageNum] = useState(1);
    64→    64→  const [textItems, setTextItems] = useState<TextItem[]>([]);
    65→    65→  const [strokes, setStrokes] = useState<DrawStroke[]>([]);
    66→    66→  const [tool2, setTool2] = useState<"select" | "text" | "draw">("select");
    67→    67→  const [penColor, setPenColor] = useState("var(--brand)");
    68→    68→  const [penWidth, setPenWidth] = useState(3);
    69→    69→  const [textDraft, setTextDraft] = useState<{ x: number; y: number; pageIndex: number; value: string; size: number; color: [number, number, number] } | null>(null);
    70→    70→  const [history2, setHistory2] = useState<Uint8Array[]>([]);
    71→    71→  const [historyIdx, setHistoryIdx] = useState(-1);
    72→    72→  const drawCanvasRef = useRef<HTMLCanvasElement | null>(null);
    73→    73→  const drawingRef = useRef<DrawStroke | null>(null);
    74→    74→
    75→    75→  const target = sourceFiles.find((f) => f.included) ?? sourceFiles[0];
    76→    76→
    77→    77→  useEffect(() => {
    78→    78→    if (target) {
    79→    79→      // Defer setState to a microtask so we don't run it synchronously in the effect body
    80→    80→      queueMicrotask(() => {
    81→    81→        setLiveBlob(target.file);
    82→    82→      });
    83→    83→      // Prime history with original
    84→    84→      void (async () => {
    85→    85→        const bytes = new Uint8Array(await target.file.arrayBuffer());
    86→    86→        setHistory2([bytes]);
    87→    87→        setHistoryIdx(0);
    88→    88→      })();
    89→    89→    }
    90→    90→  }, [target]);
    91→    91→
    92→    92→  const updateBlob = (bytes: Uint8Array) => {
    93→    93→    const blob = new Blob([bytes as unknown as BlobPart], { type: "application/pdf" });
    94→    94→    setLiveBlob(blob);
    95→    95→    // Truncate any history forward of current index, push new state
    96→    96→    setHistory2((h) => {
    97→    97→      const trimmed = h.slice(0, historyIdx + 1);
    98→    98→      trimmed.push(bytes);
    99→    99→      return trimmed.slice(-20);
   100→   100→    });
   101→   101→    setHistoryIdx((i) => Math.min(i + 1, 19));
   102→   102→  };
   103→   103→
   104→   104→  const undo = () => {
   105→   105→    if (historyIdx <= 0) return;
   106→   106→    const newIdx = historyIdx - 1;
   107→   107→    setHistoryIdx(newIdx);
   108→   108→    const bytes = history2[newIdx];
   109→   109→    setLiveBlob(new Blob([bytes as unknown as BlobPart], { type: "application/pdf" }));
   110→   110→  };
   111→   111→  const redo = () => {
   112→   112→    if (historyIdx >= history2.length - 1) return;
   113→   113→    const newIdx = historyIdx + 1;
   114→   114→    setHistoryIdx(newIdx);
   115→   115→    const bytes = history2[newIdx];
   116→   116→    setLiveBlob(new Blob([bytes as unknown as BlobPart], { type: "application/pdf" }));
   117→   117→  };
   118→   118→
   119→   119→  if (!target || !liveBlob) {
   120→   120→    return (
   121→   121→      <ToolPageShell tool={tool} ctaLabel="" ctaDisabled>
   122→   122→        <div className="mt-5 rounded-xl border border-dashed border-[var(--border)] bg-[var(--muted)] p-6 text-center text-sm text-[var(--muted-foreground)]">
   123→   123→          Drop a PDF above to start editing.
   124→   124→        </div>
   125→   125→      </ToolPageShell>
   126→   126→    );
   127→   127→  }
   128→   128→
   129→   129→  const rotate = async (degrees: 90 | 180 | 270) => {
   130→   130→    try {
   131→   131→      startProgress(`Rotating page ${pageNum}…`, "determinate", 0);
   132→   132→      const out = await rotatePdfPage(liveBlob, pageNum - 1, degrees, (pct, msg) => updateProgress(msg, pct));
   133→   133→      updateBlob(out);
   134→   134→      stopProgress();
   135→   135→      toast.success(`Rotated page ${pageNum} by ${degrees}° (other pages unchanged)`);
   136→   136→    } catch (e) {
   137→   137→      stopProgress();
   138→   138→      toast.error(e instanceof Error ? e.message : "Rotate failed");
   139→   139→    }
   140→   140→  };
   141→   141→
   142→   142→  const rotateAll = async (degrees: 90 | 180 | 270) => {
   143→   143→    try {
   144→   144→      startProgress(`Rotating every page ${degrees}°…`, "determinate", 0);
   145→   145→      const out = await rotateAllPages(liveBlob, degrees, (pct, msg) => updateProgress(msg, pct));
   146→   146→      updateBlob(out);
   147→   147→      stopProgress();
   148→   148→      toast.success(`Rotated every page by ${degrees}°`);
   149→   149→    } catch (e) {
   150→   150→      stopProgress();
   151→   151→      toast.error(e instanceof Error ? e.message : "Rotate all failed");
   152→   152→    }
   153→   153→  };
   154→   154→
   155→   155→  const del = async () => {
   156→   156→    if (!confirm(`Delete page ${pageNum}? This cannot be undone via this dialog (but you can press Undo).`)) return;
   157→   157→    try {
   158→   158→      startProgress(`Deleting page ${pageNum}…`, "determinate", 0);
   159→   159→      const out = await deletePdfPage(liveBlob, pageNum - 1, (pct, msg) => updateProgress(msg, pct));
   160→   160→      updateBlob(out);
   161→   161→      stopProgress();
   162→   162→      toast.success(`Deleted page ${pageNum}`);
   163→   163→    } catch (e) {
   164→   164→      stopProgress();
   165→   165→      toast.error(e instanceof Error ? e.message : "Delete failed");
   166→   166→    }
   167→   167→  };
   168→   168→
   169→   169→  const duplicate = async () => {
   170→   170→    try {
   171→   171→      startProgress(`Duplicating page ${pageNum}…`, "determinate", 0);
   172→   172→      const out = await duplicatePdfPage(liveBlob, pageNum - 1, (pct, msg) => updateProgress(msg, pct));
   173→   173→      updateBlob(out);
   174→   174→      stopProgress();
   175→   175→      toast.success(`Duplicated page ${pageNum}`);
   176→   176→    } catch (e) {
   177→   177→      stopProgress();
   178→   178→      toast.error(e instanceof Error ? e.message : "Duplicate failed");
   179→   179→    }
   180→   180→  };
   181→   181→
   182→   182→  const insertBlank = async () => {
   183→   183→    try {
   184→   184→      startProgress(`Inserting blank page after ${pageNum}…`, "determinate", 0);
   185→   185→      const out = await insertBlankPage(liveBlob, pageNum, (pct, msg) => updateProgress(msg, pct));
   186→   186→      updateBlob(out);
   187→   187→      stopProgress();
   188→   188→      toast.success(`Inserted blank page`);
   189→   189→      setPageNum((p) => p + 1);
   190→   190→    } catch (e) {
   191→   191→      stopProgress();
   192→   192→      toast.error(e instanceof Error ? e.message : "Insert failed");
   193→   193→    }
   194→   194→  };
   195→   195→
   196→   196→  const save = async () => {
   197→   197→    if (!liveBlob) return;
   198→   198→    // Bake any pending text items onto their pages
   199→   199→    let blob = liveBlob;
   200→   200→    const pending = textItems.filter((t) => !t.baked);
   201→   201→    for (const t of pending) {
   202→   202→      try {
   203→   203→        startProgress(`Baking text on page ${t.pageIndex + 1}…`);
   204→   204→        const out = await addTextToPage(blob, t.pageIndex, t.text, {
   205→   205→          x: t.x,
   206→   206→          y: t.y,
   207→   207→          size: t.size,
   208→   208→          color: t.color,
   209→   209→          font: "Helvetica",
   210→   210→        });
   211→   211→        blob = new Blob([out as unknown as BlobPart], { type: "application/pdf" });
   212→   212→      } catch (e) {
   213→   213→        // ignore single-page errors
   214→   214→      }
   215→   215→    }
   216→   216→    // Bake strokes
   217→   217→    for (const s of strokes) {
   218→   218→      try {
   219→   219→        // Rasterize stroke to a transparent PNG + its bounding box (in PDF points).
   220→   220→        const strokePng = await strokeToPng(s);
   221→   221→        if (strokePng.bytes.length === 0) continue;
   222→   222→        startProgress(`Baking drawing on page ${s.pageIndex + 1}…`);
   223→   223→        // Embed at the stroke's REAL bounding-box position + size (NOT full A4).
   224→   224→        // embedStrokeOnPage handles the Y-axis flip (screen top-down → PDF bottom-up).
   225→   225→        const out = await embedStrokeOnPage(blob, s.pageIndex, strokePng);
   226→   226→        blob = new Blob([out as unknown as BlobPart], { type: "application/pdf" });
   227→   227→      } catch (e) {
   228→   228→        // ignore
   229→   229→      }
   230→   230→    }
   231→   231→    setLiveBlob(blob);
   232→   232→    setTextItems((items) => items.map((t) => ({ ...t, baked: true })));
   233→   233→    setStrokes([]);
   234→   234→    stopProgress();
   235→   235→    setResult({
   236→   236→      blob,
   237→   237→      name: withExt(target.name, "-edited.pdf"),
   238→   238→      type: "application/pdf",
   239→   239→      ext: ".pdf",
   240→   240→      size: blob.size,
   241→   241→      beforeSize: target.size,
   242→   242→      beforePreviewUrl: makePreviewUrl(target.file),
   243→   243→    });
   244→   244→    setView("result");
   245→   245→    toast.success("Saved — review the result");
   246→   246→  };
   247→   247→
   248→   248→  // Click handler: when text tool active, click on canvas drops a text box
   249→   249→  const onCanvasClick = (pageIndex: number, xRatio: number, yRatio: number) => {
   250→   250→    if (tool2 !== "text" || !liveBlob) return;
   251→   251→    // Get page dimensions (approximate A4 portrait)
   252→   252→    // We'll defer to actual page size via PdfPreview's overlay; use ratios
   253→   253→    // Use 595x842 fallback for the PDF coordinate space.
   254→   254→    const pageW = 595.28;
   255→   255→    const pageH = 841.89;
   256→   256→    const x = xRatio * pageW;
   257→   257→    const y = yRatio * pageH;
   258→   258→    setTextDraft({ x, y, pageIndex, value: "", size: 14, color: [0.12, 0.15, 0.2] });
   259→   259→  };
   260→   260→
   261→   261→  const commitTextDraft = () => {
   262→   262→    if (!textDraft || !textDraft.value.trim()) {
   263→   263→      setTextDraft(null);
   264→   264→      return;
   265→   265→    }
   266→   266→    const item: TextItem = {
   267→   267→      id: `t${Date.now()}`,
   268→   268→      pageIndex: textDraft.pageIndex,
   269→   269→      x: textDraft.x,
   270→   270→      y: textDraft.y,
   271→   271→      text: textDraft.value,
   272→   272→      size: textDraft.size,
   273→   273→      color: textDraft.color,
   274→   274→      baked: false,
   275→   275→    };
   276→   276→    setTextItems((items) => [...items, item]);
   277→   277→    setTextDraft(null);
   278→   278→  };
   279→   279→
   280→   280→  return (
   281→   281→    <ToolPageShell tool={tool} ctaLabel="Save changes" ctaColor="var(--cat-edit)" onCtaClick={save}>
   282→   282→      <div className="mt-5 flex flex-col gap-3 sm:flex-row" style={{ height: "calc(100vh - 200px)", minHeight: "350px" }}>
   283→   283→        {/* Sidebar — vertical on desktop, horizontal scroll on mobile */}
   284→   284→        <div
   285→   285→          className="flex flex-row gap-1 overflow-x-auto rounded-xl border border-[var(--border)] bg-[var(--card)] p-1.5 shadow-sm thin-scroll sm:flex-col sm:w-16 sm:overflow-y-auto sm:overflow-x-hidden shrink-0"
   286→   286→        >
   287→   287→          <ToolButton onClick={undo} disabled={historyIdx <= 0} label="Undo last change" desc="Step back one change. Other pages stay the same.">
   288→   288→            <Undo2 className="size-4" />
   289→   289→          </ToolButton>
   290→   290→          <ToolButton onClick={redo} disabled={historyIdx >= history2.length - 1} label="Redo" desc="Re-apply a change you just undid.">
   291→   291→            <Redo2 className="size-4" />
   292→   292→          </ToolButton>
   293→   293→          <div className="mx-1 h-8 w-px shrink-0 bg-[var(--border)] sm:mx-0 sm:my-1 sm:h-px sm:w-full" />
   294→   294→          <ToolButton onClick={() => setTool2("select")} active={tool2 === "select"} label="Select" desc="Move around without drawing anything.">
   295→   295→            <MousePointer2 className="size-4" />
   296→   296→          </ToolButton>
   297→   297→          <ToolButton onClick={() => setTool2("text")} active={tool2 === "text"} label="Add text" desc="Add text anywhere on the page.">
   298→   298→            <Type className="size-4" />
   299→   299→          </ToolButton>
   300→   300→          <ToolButton onClick={() => setTool2("draw")} active={tool2 === "draw"} label="Draw freehand" desc="Draw freehand on the page — strokes are baked in on Save.">
   301→   301→            <PenLine className="size-4" />
   302→   302→          </ToolButton>
   303→   303→          <div className="mx-1 h-8 w-px shrink-0 bg-[var(--border)] sm:mx-0 sm:my-1 sm:h-px sm:w-full" />
   304→   304→          <ToolButton onClick={() => rotate(90)} label={`Rotate page ${pageNum} clockwise`} desc={`Rotate this page clockwise (other pages stay the same)`}>
   305→   305→            <RotateCw className="size-4" />
   306→   306→          </ToolButton>
   307→   307→          <ToolButton onClick={() => rotate(270)} label={`Rotate page ${pageNum} counter-clockwise`} desc={`Rotate this page counter-clockwise (other pages stay the same)`}>
   308→   308→            <RotateCcw className="size-4" />
   309→   309→          </ToolButton>
   310→   310→          <ToolButton onClick={() => rotateAll(90)} label="Rotate ALL pages 90° clockwise" desc="Rotate every page in this PDF by 90° clockwise — a bulk action.">
   311→   311→            <span className="relative">
   312→   312→              <RotateCw className="size-4" />
   313→   313→              <span className="absolute -right-2 -top-2 rounded bg-[var(--cat-edit)] px-1 py-0 text-[8px] font-bold leading-tight text-white">ALL</span>
   314→   314→            </span>
   315→   315→          </ToolButton>
   316→   316→          <ToolButton onClick={duplicate} label={`Duplicate page ${pageNum}`} desc={`Make a copy of this page right after it.`}>
   317→   317→            <Copy className="size-4" />
   318→   318→          </ToolButton>
   319→   319→          <ToolButton onClick={insertBlank} label={`Insert blank page after ${pageNum}`} desc={`Add an empty page right after this one.`}>
   320→   320→            <Plus className="size-4" />
   321→   321→          </ToolButton>
   322→   322→          <ToolButton onClick={del} label={`Delete page ${pageNum}`} desc={`Remove this page from the PDF.`} danger>
   323→   323→            <Trash2 className="size-4" />
   324→   324→          </ToolButton>
   325→   325→
   326→   326→          {/* Draw toolbar — inline in sidebar when draw is active */}
   327→   327→          {tool2 === "draw" && (
   328→   328→            <>
   329→   329→              <div className="mx-1 h-8 w-px shrink-0 bg-[var(--border)] sm:mx-0 sm:my-1 sm:h-px sm:w-full" />
   330→   330→              {["var(--foreground)", "var(--brand)", "var(--danger)", "var(--success)", "var(--cat-convert)"].map((c) => (
   331→   331→                <button
   332→   332→                  key={c}
   333→   333→                  onClick={() => setPenColor(c)}
   334→   334→                  className={cn("size-7 rounded-full border-2 transition-all hover:scale-110", penColor === c ? "border-[var(--brand)] scale-110" : "border-[var(--border)]")}
   335→   335→                  style={{ background: c }}
   336→   336→                  aria-label={`Pen color ${c}`}
   337→   337→                />
   338→   338→              ))}
   339→   339→              <input
   340→   340→                type="range"
   341→   341→                min="1"
   342→   342→                max="10"
   343→   343→                step="0.5"
   344→   344→                value={penWidth}
   345→   345→                onChange={(e) => setPenWidth(parseFloat(e.target.value))}
   346→   346→                className="w-12 accent-[var(--cat-edit)]"
   347→   347→                title="Pen width"
   348→   348→              />
   349→   349→              <button
   350→   350→                onClick={() => setStrokes((s) => s.filter((st) => st.pageIndex !== pageNum - 1))}
   351→   351→                className="rounded-md border border-[var(--border)] px-2 py-1 text-[10px] text-[var(--muted-foreground)] hover:bg-[var(--muted)]"
   352→   352→                title="Clear strokes on this page"
   353→   353→              >
   354→   354→                Clear
   355→   355→              </button>
   356→   356→            </>
   357→   357→          )}
   358→   358→
   359→   359→          <div className="mt-auto pt-2">
   360→   360→            <ToolButton onClick={save} label="Save and review the result" desc="Bake in all changes and go to the result screen." cta>
   361→   361→              <Save className="size-4" />
   362→   362→            </ToolButton>
   363→   363→          </div>
   364→   364→        </div>
   365→   365→
   366→   366→        {/* Large PDF preview — takes the rest of the space */}
   367→   367→        <div className="flex-1 min-w-0 overflow-hidden rounded-2xl border border-[var(--border)] bg-[var(--card)] shadow-sm">
   368→   368→          <div className="relative h-full">
   369→   369→            <PdfPreview
   370→   370→              blob={liveBlob}
   371→   371→              initialScale={0}
   372→   372→              onCanvasClick={onCanvasClick}
   373→   373→              renderOverlay={(pageIndex, pageW, pageH, scale) => (
   374→   374→                <EditOverlay
   375→   375→                  pageIndex={pageIndex}
   376→   376→                  pageW={pageW}
   377→   377→                  pageH={pageH}
   378→   378→                  scale={scale}
   379→   379→                  pageNum={pageNum}
   380→   380→                  setPageNum={setPageNum}
   381→   381→                  textItems={textItems}
   382→   382→                  strokes={strokes}
   383→   383→                  setStrokes={setStrokes}
   384→   384→                  tool2={tool2}
   385→   385→                  penColor={penColor}
   386→   386→                  penWidth={penWidth}
   387→   387→                  textDraft={textDraft}
   388→   388→                  setTextDraft={setTextDraft}
   389→   389→                  commitTextDraft={commitTextDraft}
   390→   390→                  setTextItems={setTextItems}
   391→   391→                />
   392→   392→              )}
   393→   393→            />
   394→   394→          </div>
   395→   395→        </div>
   396→   396→      </div>
   397→   397→
   398→   398→      {/* Help note — BELOW the sidebar+preview row (full width).
   399→   399→          Previously this sat INSIDE the sm:flex-row as a third flex item, where
   400→   400→          its long text ate ~1326px and left only ~2px for the preview, so the
   401→   401→          PDF canvas Fit-computed to a 30px-wide sliver ("picture not there"). */}
   402→   402→      <div className="mt-3 rounded-lg border border-[var(--border)] bg-[var(--muted)] p-3 text-xs text-[var(--muted-foreground)]">
   403→   403→        <b className="text-[var(--foreground)]">Per-page rotation:</b> rotate / delete / duplicate buttons above operate on the
   404→   404→        page currently in view only — other pages are left untouched. Use the page navigation arrows in the preview
   405→   405→        toolbar to switch pages, or the +/- zoom controls.
   406→   406→      </div>
   407→   407→    </ToolPageShell>
   408→   408→  );
   409→   409→}
   410→   410→
   411→   411→function EditOverlay(props: {
   412→   412→  pageIndex: number;
   413→   413→  pageW: number;
   414→   414→  pageH: number;
   415→   415→  scale: number;
   416→   416→  pageNum: number;
   417→   417→  setPageNum: (n: number | ((n: number) => number)) => void;
   418→   418→  textItems: TextItem[];
   419→   419→  strokes: DrawStroke[];
   420→   420→  setStrokes: React.Dispatch<React.SetStateAction<DrawStroke[]>>;
   421→   421→  tool2: "select" | "text" | "draw";
   422→   422→  penColor: string;
   423→   423→  penWidth: number;
   424→   424→  textDraft: { x: number; y: number; pageIndex: number; value: string; size: number; color: [number, number, number] } | null;
   425→   425→  setTextDraft: React.Dispatch<React.SetStateAction<{ x: number; y: number; pageIndex: number; value: string; size: number; color: [number, number, number] } | null>>;
   426→   426→  commitTextDraft: () => void;
   427→   427→  setTextItems: React.Dispatch<React.SetStateAction<TextItem[]>>;
   428→   428→}) {
   429→   429→  const { pageIndex, pageW, pageH, scale, pageNum, setPageNum, textItems, strokes, setStrokes, tool2, penColor, penWidth, textDraft, setTextDraft, commitTextDraft, setTextItems } = props;
   430→   430→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   431→   431→  const drawingRef = useRef<DrawStroke | null>(null);
   432→   432→
   433→   433→  // Sync pageNum with the page being rendered (so toolbar actions target the right page)
   434→   434→  useEffect(() => {
   435→   435→    setPageNum(pageIndex + 1);
   436→   436→  }, [pageIndex, setPageNum]);
   437→   437→
   438→   438→  // Redraw strokes for this page
   439→   439→  useEffect(() => {
   440→   440→    const canvas = canvasRef.current;
   441→   441→    if (!canvas) return;
   442→   442→    canvas.width = pageW * scale;
   443→   443→    canvas.height = pageH * scale;
   444→   444→    const ctx = canvas.getContext("2d");
   445→   445→    if (!ctx) return;
   446→   446→    ctx.clearRect(0, 0, canvas.width, canvas.height);
   447→   447→    ctx.lineCap = "round";
   448→   448→    ctx.lineJoin = "round";
   449→   449→    for (const s of strokes) {
   450→   450→      // Defensive: never crash on a malformed stroke entry.
   451→   451→      if (!s || typeof s.pageIndex !== "number" || !Array.isArray(s.points) || s.points.length === 0) continue;
   452→   452→      if (s.pageIndex !== pageIndex) continue;
   453→   453→      ctx.strokeStyle = s.color;
   454→   454→      ctx.lineWidth = s.width * scale;
   455→   455→      ctx.beginPath();
   456→   456→      const pts = s.points;
   457→   457→      ctx.moveTo(pts[0].x * scale, pts[0].y * scale);
   458→   458→      for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x * scale, pts[i].y * scale);
   459→   459→      ctx.stroke();
   460→   460→    }
   461→   461→  }, [strokes, pageIndex, scale, pageW, pageH]);
   462→   462→
   463→   463→  const onPointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   464→   464→    if (tool2 !== "draw") return;
   465→   465→    e.preventDefault();
   466→   466→    const rect = canvasRef.current!.getBoundingClientRect();
   467→   467→    const x = (e.clientX - rect.left) / scale;
   468→   468→    const y = (e.clientY - rect.top) / scale;
   469→   469→    drawingRef.current = { pageIndex, points: [{ x, y }], color: penColor, width: penWidth };
   470→   470→    canvasRef.current?.setPointerCapture(e.pointerId);
   471→   471→  };
   472→   472→  const onPointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
   473→   473→    if (tool2 !== "draw" || !drawingRef.current) return;
   474→   474→    const rect = canvasRef.current!.getBoundingClientRect();
   475→   475→    const x = (e.clientX - rect.left) / scale;
   476→   476→    const y = (e.clientY - rect.top) / scale;
   477→   477→    drawingRef.current.points.push({ x, y });
   478→   478→    const canvas = canvasRef.current!;
   479→   479→    const ctx = canvas.getContext("2d")!;
   480→   480→    ctx.strokeStyle = drawingRef.current.color;
   481→   481→    ctx.lineWidth = drawingRef.current.width * scale;
   482→   482→    ctx.lineCap = "round";
   483→   483→    ctx.lineJoin = "round";
   484→   484→    const pts = drawingRef.current.points;
   485→   485→    ctx.beginPath();
   486→   486→    ctx.moveTo(pts[pts.length - 2].x * scale, pts[pts.length - 2].y * scale);
   487→   487→    ctx.lineTo(pts[pts.length - 1].x * scale, pts[pts.length - 1].y * scale);
   488→   488→    ctx.stroke();
   489→   489→  };
   490→   490→  const onPointerUp = () => {
   491→   491→    if (tool2 !== "draw" || !drawingRef.current) return;
   492→   492→    // Capture the stroke by VALUE before clearing the ref, because the
   493→   493→    // setStrokes updater runs asynchronously and would otherwise read null.
   494→   494→    const stroke = drawingRef.current;
   495→   495→    if (stroke.points.length > 1) {
   496→   496→      setStrokes((s) => [...s, stroke]);
   497→   497→    }
   498→   498→    drawingRef.current = null;
   499→   499→  };
   500→   500→
   501→   501→  return (
   502→   502→    <div className="absolute inset-0 pointer-events-auto" style={{ width: pageW * scale, height: pageH * scale }}>
   503→   503→      <canvas
   504→   504→        ref={canvasRef}
   505→   505→        onPointerDown={onPointerDown}
   506→   506→        onPointerMove={onPointerMove}
   507→   507→        onPointerUp={onPointerUp}
   508→   508→        className={cn("absolute inset-0", tool2 === "draw" ? "cursor-crosshair" : "cursor-default")}
   509→   509→        style={{ width: pageW * scale, height: pageH * scale }}
   510→   510→      />
   511→   511→      {/* Existing text items for this page */}
   512→   512→      {textItems
   513→   513→        .filter((t) => t.pageIndex === pageIndex)
   514→   514→        .map((t) => (
   515→   515→          <div
   516→   516→            key={t.id}
   517→   517→            className="absolute select-none rounded border border-[var(--brand)]/40 bg-[var(--card)]/80 px-1 text-[var(--foreground)]"
   518→   518→            style={{
   519→   519→              left: t.x * scale,
   520→   520→              top: t.y * scale,
   521→   521→              fontSize: t.size * scale,
   522→   522→              color: `rgb(${t.color.map((c) => Math.round(c * 255)).join(",")})`,
   523→   523→            }}
   524→   524→          >
   525→   525→            {t.text}
   526→   526→            {t.baked && <span className="ml-1 text-[8px] text-[var(--success)]">baked</span>}
   527→   527→            {!t.baked && (
   528→   528→              <button
   529→   529→                onClick={() => setTextItems((items) => items.filter((x) => x.id !== t.id))}
   530→   530→                className="ml-1 text-[10px] text-[var(--danger)]"
   531→   531→              >
   532→   532→                ×
   533→   533→              </button>
   534→   534→            )}
   535→   535→          </div>
   536→   536→        ))}
   537→   537→      {/* Draft text input */}
   538→   538→      {textDraft && textDraft.pageIndex === pageIndex && (
   539→   539→        <div
   540→   540→          className="absolute"
   541→   541→          style={{ left: textDraft.x * scale, top: textDraft.y * scale }}
   542→   542→        >
   543→   543→          <textarea
   544→   544→            autoFocus
   545→   545→            value={textDraft.value}
   546→   546→            onChange={(e) => setTextDraft((d) => (d ? { ...d, value: e.target.value } : d))}
   547→   547→            onBlur={commitTextDraft}
   548→   548→            onKeyDown={(e) => {
   549→   549→              if (e.key === "Enter" && !e.shiftKey) {
   550→   550→                e.preventDefault();
   551→   551→                commitTextDraft();
   552→   552→              }
   553→   553→              if (e.key === "Escape") {
   554→   554→                setTextDraft(null);
   555→   555→              }
   556→   556→            }}
   557→   557→            placeholder="Type text… (Enter to add, Esc to cancel)"
   558→   558→            className="min-w-[120px] rounded border border-[var(--brand)] bg-[var(--card)] px-1 py-0.5 text-sm text-[var(--foreground)] outline-none"
   559→   559→            style={{
   560→   560→              fontSize: textDraft.size * scale,
   561→   561→              color: `rgb(${textDraft.color.map((c) => Math.round(c * 255)).join(",")})`,
   562→   562→            }}
   563→   563→          />
   564→   564→        </div>
   565→   565→      )}
   566→   566→    </div>
   567→   567→  );
   568→   568→}

   570→   570→async function strokeToPng(stroke: DrawStroke): Promise<{ bytes: Uint8Array; minX: number; minY: number; maxX: number; maxY: number; w: number; h: number; pad: number }> {
   571→   571→  // Rasterize the stroke to a transparent PNG sized to the stroke's bounding box + padding.
   572→   572→  // Returns the PNG bytes AND the bounding box (in PDF points) so the embed step can
   573→   573→  // place the image at the stroke's REAL position/size (not stretched to full A4).
   574→   574→  const pts = stroke.points;
   575→   575→  if (pts.length === 0) return { bytes: new Uint8Array(0), minX: 0, minY: 0, maxX: 0, maxY: 0, w: 1, h: 1, pad: 0 };
   576→   576→  const xs = pts.map((p) => p.x);
   577→   577→  const ys = pts.map((p) => p.y);
   578→   578→  const minX = Math.min(...xs);
   579→   579→  const maxX = Math.max(...xs);
   580→   580→  const minY = Math.min(...ys);
   581→   581→  const maxY = Math.max(...ys);
   582→   582→  const pad = stroke.width + 2;
   583→   583→  const w = Math.max(1, Math.ceil(maxX - minX + pad * 2));
   584→   584→  const h = Math.max(1, Math.ceil(maxY - minY + pad * 2));
   585→   585→  // Render at 2x DPI for crisp output, but embed at the 1:1 PDF-point size (w×h)
   586→   586→  // so the stroke stays at its true dimensions — a small line stays small.
   587→   587→  const dpi = 2;
   588→   588→  const canvas = document.createElement("canvas");
   589→   589→  canvas.width = w * dpi;
   590→   590→  canvas.height = h * dpi;
   591→   591→  const ctx = canvas.getContext("2d");
   592→   592→  if (!ctx) return { bytes: new Uint8Array(0), minX, minY, maxX, maxY, w, h, pad };
   593→   593→  ctx.scale(dpi, dpi);
   594→   594→  ctx.clearRect(0, 0, w, h);
   595→   595→  ctx.lineCap = "round";
   596→   596→  ctx.lineJoin = "round";
   597→   597→  ctx.strokeStyle = stroke.color;
   598→   598→  ctx.lineWidth = stroke.width;
   599→   599→  ctx.beginPath();
   600→   600→  ctx.moveTo(pts[0].x - minX + pad, pts[0].y - minY + pad);
   601→   601→  for (let i = 1; i < pts.length; i++) {
   602→   602→    ctx.lineTo(pts[i].x - minX + pad, pts[i].y - minY + pad);
   603→   603→  }
   604→   604→  ctx.stroke();
   605→   605→  const blob: Blob = await new Promise((resolve) => canvas.toBlob((b) => resolve(b as Blob), "image/png"));
   606→   606→  const arr = new Uint8Array(await blob.arrayBuffer());
   607→   607→  return { bytes: arr, minX, minY, maxX, maxY, w, h, pad };
   608→   608→}
