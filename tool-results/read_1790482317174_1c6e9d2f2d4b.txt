    76→function drawStroke(ctx: CanvasRenderingContext2D, s: Stroke) {
    77→  if (s.points.length === 0) return;
    78→  ctx.save();
    79→  if (s.kind === "highlighter") {
    80→    ctx.globalAlpha = 0.35;
    81→    ctx.globalCompositeOperation = "multiply";
    82→  }
    83→  ctx.strokeStyle = s.color;
    84→  ctx.lineWidth = s.width;
    85→  ctx.lineCap = "round";
    86→  ctx.lineJoin = "round";
    87→  ctx.beginPath();
    88→  ctx.moveTo(s.points[0].x, s.points[0].y);
    89→  for (let i = 1; i < s.points.length; i++) {
    90→    ctx.lineTo(s.points[i].x, s.points[i].y);
    91→  }
    92→  ctx.stroke();
    93→  ctx.restore();
    94→}

   100→  // been left in the persistent session from another tool.
   101→  const target =
   102→    sourceFiles.find((f) => f.included && (isPng(f.file) || isJpg(f.file))) ??
   103→    sourceFiles.find((f) => isPng(f.file) || isJpg(f.file));
   104→
   105→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   106→  const overlayRef = useRef<HTMLDivElement | null>(null);
   107→  const imgElRef = useRef<HTMLImageElement | null>(null);
   108→  const drawingRef = useRef(false);
   109→  const draftStartRef = useRef<{ x: number; y: number } | null>(null);
   110→
   111→  const [imageLoaded, setImageLoaded] = useState(false);
   112→  const [imageDims, setImageDims] = useState<{ w: number; h: number } | null>(null);
   113→  const [imageFormat, setImageFormat] = useState<"png" | "jpg">("png");
   114→  const [rotation, setRotation] = useState(0); // 0/90/180/270
   115→  const [tool2, setTool2] = useState<ToolKind>("select");
   116→  const [penColor, setPenColor] = useState(PEN_COLORS[0]);
   117→  const [penWidth, setPenWidth] = useState(4);
   118→  const [strokes, setStrokes] = useState<Stroke[]>([]);
   119→  const [shapes, setShapes] = useState<Shape[]>([]);
   120→  const [texts, setTexts] = useState<TextItem[]>([]);
   121→  const [draft, setDraft] = useState<Stroke | Shape | null>(null);
   122→  const [textDraft, setTextDraft] = useState<{
   123→    x: number;
   124→    y: number;
   125→    value: string;
   126→  } | null>(null);
   127→  const [editingTextId, setEditingTextId] = useState<string | null>(null);
   128→  const [undoStack, setUndoStack] = useState<Snapshot[]>([]);
   129→  const [redoStack, setRedoStack] = useState<Snapshot[]>([]);
   130→  const [busy, setBusy] = useState(false);
   131→
   132→  // Load image as HTMLImageElement for rendering to canvas.
   133→  useEffect(() => {
   134→    if (!target) {
   135→      setImageLoaded(false);
   136→      setImageDims(null);
   137→      return;
   138→    }
   139→    const img = new Image();
   140→    img.onload = () => {
   141→      imgElRef.current = img;
   142→      setImageDims({ w: img.naturalWidth, h: img.naturalHeight });
   143→      setImageFormat(isPng(target.file) ? "png" : "jpg");
   144→      setImageLoaded(true);
   145→      setRotation(0);
   146→      setStrokes([]);
   147→      setShapes([]);
   148→      setTexts([]);
   149→      setUndoStack([]);
   150→      setRedoStack([]);
   151→    };
   152→    img.onerror = () => {
   153→      toast.error("Could not load image.");
   154→    };
   155→    img.src = URL.createObjectURL(target.file);
   156→    return () => {
   157→      URL.revokeObjectURL(img.src);
   158→      imgElRef.current = null;
   159→    };
   160→  }, [target]);
   161→
   162→  // Displayed canvas size — fit within a max viewport while preserving
   163→  // aspect ratio. Account for rotation (swap dims at 90/270).
   164→  const [displaySize, setDisplaySize] = useState<{ w: number; h: number } | null>(null);
   165→
   166→  useEffect(() => {
   167→    if (!imageDims) return;
   168→    const rotated = rotation === 90 || rotation === 270;
   169→    const baseW = rotated ? imageDims.h : imageDims.w;
   170→    const baseH = rotated ? imageDims.w : imageDims.h;
   171→    const maxW = 900;
   172→    const maxH = 600;
   173→    const scale = Math.min(maxW / baseW, maxH / baseH, 1);
   174→    setDisplaySize({
   175→      w: Math.round(baseW * scale),
   176→      h: Math.round(baseH * scale),
   177→    });
   178→  }, [imageDims, rotation]);
   179→
   180→  // The conversion factor from canvas pixel coords → original image pixels
   181→  // (for crop & save).
   182→  const getScale = useCallback(() => {
   183→    if (!imageDims || !displaySize) return 1;
   184→    const rotated = rotation === 90 || rotation === 270;
   185→    const baseW = rotated ? imageDims.h : imageDims.w;
   186→    return baseW / displaySize.w;
   187→  }, [imageDims, displaySize, rotation]);
   188→
   189→  // Re-render the canvas: image + annotations + draft.
   190→  const renderCanvas = useCallback(() => {
   191→    const canvas = canvasRef.current;
   192→    const img = imgElRef.current;
   193→    if (!canvas || !img || !displaySize) return;
   194→    canvas.width = displaySize.w;
   195→    canvas.height = displaySize.h;
   196→    const ctx = canvas.getContext("2d");
   197→    if (!ctx) return;
   198→
   199→    // Clear & paint background (white — JPGs may have alpha).
   200→    ctx.fillStyle = "#FFFFFF";
   201→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   202→
   203→    // Draw the image with rotation. We translate to center, rotate,
   204→    // then draw image centered.
   205→    ctx.save();
   206→    ctx.translate(canvas.width / 2, canvas.height / 2);
   207→    ctx.rotate((rotation * Math.PI) / 180);
   208→    const rotated = rotation === 90 || rotation === 270;
   209→    const drawW = rotated ? canvas.height : canvas.width;
   210→    const drawH = rotated ? canvas.width : canvas.height;
   211→    ctx.drawImage(img, -drawW / 2, -drawH / 2, drawW, drawH);
   212→    ctx.restore();
   213→
   214→    // Draw shapes (under strokes for cleaner layering).
   215→    for (const s of shapes) {
   216→      ctx.save();
   217→      if (s.kind === "rect") {
   218→        ctx.strokeStyle = s.color;
   219→        ctx.lineWidth = s.w === 0 ? 1 : 2; // not used; use penWidth style
   220→        ctx.lineWidth = penWidth;
   221→        if (s.fill) {
   222→          ctx.fillStyle = s.color;
   223→          ctx.fillRect(s.x, s.y, s.w, s.h);
   224→        } else {
   225→          ctx.strokeRect(s.x, s.y, s.w, s.h);
   226→        }
   227→      } else if (s.kind === "circle") {
   228→        ctx.strokeStyle = s.color;
   229→        ctx.lineWidth = penWidth;
   230→        ctx.beginPath();
   231→        const rx = Math.abs(s.w) / 2;
   232→        const ry = Math.abs(s.h) / 2;
   233→        ctx.ellipse(
   234→          s.x + s.w / 2,
   235→          s.y + s.h / 2,
   236→          rx,
   237→          ry,
   238→          0,
   239→          0,
   240→          Math.PI * 2,
   241→        );
   242→        if (s.fill) {
   243→          ctx.fillStyle = s.color;
   244→          ctx.fill();
   245→        } else {
   246→          ctx.stroke();
   247→        }
   248→      }
   249→      ctx.restore();
   250→    }
   251→
   252→    // Draw draft shape (rectangle/circle drag).
   253→    if (draft && draft.kind === "rect") {
   254→      const d = draft as Shape;
   255→      ctx.save();
   256→      ctx.strokeStyle = d.color;
   257→      ctx.lineWidth = penWidth;
   258→      ctx.strokeRect(d.x, d.y, d.w, d.h);
   259→      ctx.restore();
   260→    } else if (draft && draft.kind === "circle") {
   261→      const d = draft as Shape;
   262→      ctx.save();
   263→      ctx.strokeStyle = d.color;
   264→      ctx.lineWidth = penWidth;
   265→      ctx.beginPath();
   266→      ctx.ellipse(
   267→        d.x + d.w / 2,
   268→        d.y + d.h / 2,
   269→        Math.abs(d.w) / 2,
   270→        Math.abs(d.h) / 2,
   271→        0,
   272→        0,
   273→        Math.PI * 2,
   274→      );
   275→      ctx.stroke();
   276→      ctx.restore();
   277→    } else if (draft && (draft.kind === "draw" || draft.kind === "highlighter")) {
   278→      drawStroke(ctx, draft as Stroke);
   279→    }
   280→
   281→    // Draw strokes.
   282→    for (const s of strokes) {
   283→      drawStroke(ctx, s);
   284→    }
   285→
   286→    // Draw text items.
   287→    for (const t of texts) {
   288→      ctx.save();
   289→      ctx.fillStyle = t.color;
   290→      ctx.font = `${t.size}px Inter, Helvetica, Arial, sans-serif`;
   291→      ctx.textBaseline = "top";
   292→      // Handle multi-line text.
   293→      const lines = t.value.split("\n");
   294→      for (let i = 0; i < lines.length; i++) {
   295→        ctx.fillText(lines[i], t.x, t.y + i * t.size * 1.2);
   296→      }
   297→      ctx.restore();
   298→    }
   299→
   300→    // Draw text draft.
   301→    if (textDraft) {
   302→      ctx.save();
   303→      ctx.fillStyle = penColor;
   304→      ctx.font = `${Math.max(14, penWidth * 4)}px Inter, Helvetica, Arial, sans-serif`;
   305→      ctx.textBaseline = "top";
   306→      ctx.fillText(textDraft.value || "", textDraft.x, textDraft.y);
   307→      ctx.restore();
   308→    }
   309→  }, [displaySize, rotation, shapes, strokes, texts, draft, textDraft, penWidth, penColor]);
   310→
   311→  // Re-render whenever state changes.
   312→  useEffect(() => {
   313→    renderCanvas();
   314→  }, [renderCanvas]);
   315→
   316→  // Push current state to undo stack, clear redo.
   317→  const snapshot = (): Snapshot => ({
   318→    strokes: strokes.map((s) => ({ ...s, points: [...s.points] })),
   319→    shapes: shapes.map((s) => ({ ...s })),
   320→    texts: texts.map((t) => ({ ...t })),
   321→    rotation,
   322→    imgWidth: imageDims?.w ?? 0,
   323→    imgHeight: imageDims?.h ?? 0,
   324→  });
   325→
   326→  const pushUndo = () => {
   327→    setUndoStack((s) => [...s.slice(-19), snapshot()]);
   328→    setRedoStack([]);
   329→  };
   330→
   331→  const undo = () => {
   332→    if (undoStack.length === 0) {
   333→      toast.info("Nothing to undo.");
   334→      return;
   335→    }
   336→    const prev = undoStack[undoStack.length - 1];
   337→    setRedoStack((s) => [...s, snapshot()]);
   338→    setStrokes(prev.strokes.map((s) => ({ ...s, points: [...s.points] })));
   339→    setShapes(prev.shapes.map((s) => ({ ...s })));
   340→    setTexts(prev.texts.map((t) => ({ ...t })));
   341→    setRotation(prev.rotation);
   342→    setUndoStack((s) => s.slice(0, -1));
   343→  };
   344→
   345→  const redo = () => {
   346→    if (redoStack.length === 0) {
   347→      toast.info("Nothing to redo.");
   348→      return;
   349→    }
   350→    const next = redoStack[redoStack.length - 1];
   351→    setUndoStack((s) => [...s, snapshot()]);
   352→    setStrokes(next.strokes.map((s) => ({ ...s, points: [...s.points] })));
   353→    setShapes(next.shapes.map((s) => ({ ...s })));
   354→    setTexts(next.texts.map((t) => ({ ...t })));
   355→    setRotation(next.rotation);
   356→    setRedoStack((s) => s.slice(0, -1));
   357→  };
   358→
   359→  // Pointer handlers — convert client coords to canvas pixel coords.
   360→  const getCanvasPos = (e: React.PointerEvent) => {
   361→    const canvas = canvasRef.current!;
   362→    const rect = canvas.getBoundingClientRect();
   363→    const x = ((e.clientX - rect.left) / rect.width) * canvas.width;
   364→    const y = ((e.clientY - rect.top) / rect.height) * canvas.height;
   365→    return { x, y };
   366→  };
   367→
   368→  const onPointerDown = (e: React.PointerEvent) => {
   369→    if (tool2 === "select") return;
   370→    const { x, y } = getCanvasPos(e);
   371→    draftStartRef.current = { x, y };
   372→
   373→    if (tool2 === "draw" || tool2 === "highlighter") {
   374→      drawingRef.current = true;
   375→      pushUndo();
   376→      setDraft({
   377→        id: `s-${Date.now()}`,
   378→        kind: tool2,
   379→        points: [{ x, y }],
   380→        color: penColor,
   381→        width: penWidth,
   382→      });
   383→    } else if (tool2 === "rect" || tool2 === "circle") {
   384→      drawingRef.current = true;
   385→      pushUndo();
   386→      setDraft({
   387→        id: `sh-${Date.now()}`,
   388→        kind: tool2,
   389→        x,
   390→        y,
   391→        w: 0,
   392→        h: 0,
   393→        color: penColor,
   394→        fill: false,
   395→      });
   396→    } else if (tool2 === "text") {
   397→      // Drop a text box if editing mode is off, otherwise commit current.
   398→      if (textDraft) {
   399→        commitTextDraft();
   400→      }
   401→      setTextDraft({ x, y, value: "" });
   402→      setEditingTextId(null);
   403→    } else if (tool2 === "crop") {
   404→      drawingRef.current = true;
   405→      setDraft({
   406→        id: `cr-${Date.now()}`,
   407→        kind: "rect",
   408→        x,
   409→        y,
   410→        w: 0,
   411→        h: 0,
   412→        color: "var(--brand)",
   413→        fill: false,
   414→      });
   415→    }
   416→    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
   417→  };
   418→
   419→  const onPointerMove = (e: React.PointerEvent) => {
   420→    if (!drawingRef.current || !draftStartRef.current) return;
   421→    const { x, y } = getCanvasPos(e);
   422→    if (tool2 === "draw" || tool2 === "highlighter") {
   423→      setDraft((d) => {
   424→        if (!d || !(d.kind === "draw" || d.kind === "highlighter")) return d;
   425→        const s = d as Stroke;
   426→        return { ...s, points: [...s.points, { x, y }] };
   427→      });
   428→    } else if (tool2 === "rect" || tool2 === "circle" || tool2 === "crop") {
   429→      const start = draftStartRef.current;
   430→      setDraft((d) => {
   431→        if (!d) return d;
   432→        const sh = d as Shape;
   433→        return {
   434→          ...sh,
   435→          x: Math.min(start.x, x),
   436→          y: Math.min(start.y, y),
   437→          w: Math.abs(x - start.x),
   438→          h: Math.abs(y - start.y),
   439→        };
   440→      });
   441→    }
   442→  };
   443→
   444→  const onPointerUp = (e: React.PointerEvent) => {
   445→    drawingRef.current = false;
   446→    (e.target as HTMLElement).releasePointerCapture?.(e.pointerId);
   447→    if (!draftStartRef.current) return;
   448→    if (tool2 === "draw" || tool2 === "highlighter") {
   449→      setDraft((d) => {
   450→        if (d && (d.kind === "draw" || d.kind === "highlighter")) {
   451→          setStrokes((prev) => [...prev, d as Stroke]);
   452→        }
   453→        return null;
   454→      });
   455→    } else if (tool2 === "rect" || tool2 === "circle") {
   456→      setDraft((d) => {
   457→        if (d && (d.kind === "rect" || d.kind === "circle")) {
   458→          const sh = d as Shape;
   459→          if (sh.w > 4 && sh.h > 4) {
   460→            setShapes((prev) => [...prev, sh]);
   461→          }
   462→        }
   463→        return null;
   464→      });
   465→    }
   466→    draftStartRef.current = null;
   467→  };
   468→
   469→  const commitTextDraft = () => {
   470→    if (!textDraft) return;
   471→    if (!textDraft.value.trim()) {
   472→      setTextDraft(null);
   473→      return;
   474→    }
   475→    pushUndo();
   476→    setTexts((prev) => [
   477→      ...prev,
   478→      {
   479→        id: `t-${Date.now()}`,
   480→        x: textDraft.x,
   481→        y: textDraft.y,
   482→        value: textDraft.value,
   483→        color: penColor,
   484→        size: Math.max(14, penWidth * 4),
   485→      },
   486→    ]);
   487→    setTextDraft(null);
   488→  };
   489→
   490→  const rotate = (direction: "cw" | "ccw") => {
   491→    pushUndo();
   492→    setRotation((r) => {
   493→      const delta = direction === "cw" ? 90 : -90;
   494→      const next = (r + delta + 360) % 360;
   495→      return next;
   496→    });
   497→  };
   498→
   499→  const applyCrop = () => {
   500→    if (!draft || draft.kind !== "rect") {
   501→      toast.error("Drag a rectangle on the image with the Crop tool first.");
   502→      return;
   503→    }
   504→    const rect = draft as Shape;
   505→    if (rect.w < 4 || rect.h < 4) {
   506→      toast.error("Crop rectangle is too small.");
   507→      return;
   508→    }
   509→    setBusy(true);
   510→    try {
   511→      const canvas = canvasRef.current!;
   512→      const scale = getScale();
   513→      const cropX = Math.max(0, Math.round(rect.x * scale));
   514→      const cropY = Math.max(0, Math.round(rect.y * scale));
   515→      const cropW = Math.max(1, Math.round(rect.w * scale));
   516→      const cropH = Math.max(1, Math.round(rect.h * scale));
   517→
   518→      // Create a new canvas with the cropped region, render to image bitmap.
   519→      const tmp = document.createElement("canvas");
   520→      tmp.width = cropW;
   521→      tmp.height = cropH;
   522→      const tctx = tmp.getContext("2d")!;
   523→      tctx.drawImage(canvas, cropX, cropY, cropW, cropH, 0, 0, cropW, cropH);
   524→      const dataUrl = tmp.toDataURL(imageFormat === "png" ? "image/png" : "image/jpeg", 0.92);
   525→
   526→      // Load the new image and replace our imgEl.
   527→      const newImg = new Image();
   528→      newImg.onload = () => {
   529→        imgElRef.current = newImg;
   530→        setImageDims({ w: cropW, h: cropH });
   531→        setStrokes([]);
   532→        setShapes([]);
   533→        setTexts([]);
   534→        setRotation(0);
   535→        setDraft(null);
   536→        setBusy(false);
   537→        toast.success(`Cropped to ${cropW}×${cropH}px`);
   538→      };
   539→      newImg.onerror = () => {
   540→        setBusy(false);
   541→        toast.error("Crop failed.");
   542→      };
   543→      newImg.src = dataUrl;
   544→    } catch (e) {
   545→      setBusy(false);
   546→      toast.error(e instanceof Error ? e.message : "Crop failed");
   547→    }
   548→  };
   549→
   550→  const save = () => {
   551→    const canvas = canvasRef.current;
   552→    if (!canvas) {
   553→      toast.error("Nothing to save yet.");
   554→      return;
   555→    }
   556→    setBusy(true);
   557→    try {
   558→      const mime = imageFormat === "png" ? "image/png" : "image/jpeg";
   559→      const quality = imageFormat === "png" ? undefined : 0.92;
   560→      canvas.toBlob(
   561→        (blob) => {
   562→          if (!blob) {
   563→            setBusy(false);
   564→            toast.error("Could not export image.");
   565→            return;
   566→          }
   567→          setResult({
   568→            blob,
   569→            name: withExt(target?.name ?? "edited", `.${imageFormat}`),
   570→            type: mime,
   571→            ext: `.${imageFormat}`,
   572→            size: blob.size,
   573→            beforeSize: target?.size,
   574→          });
   575→          setBusy(false);

   578→        },

   919→                  style={{ color: "var(--muted-foreground)" }}

   921→                  Format: {imageFormat.toUpperCase()}
   922→                  {rotation !== 0 ? ` · rotated ${rotation}°` : ""}

   934→              {tool2 === "crop" && (

   940→                    (busy || !draft) && "cursor-not-allowed opacity-50",

   947→
   948→              <button
   949→                onClick={downloadNow}
   950→                className="inline-flex w-full items-center justify-center gap-1.5 rounded-lg border px-3 py-2 text-xs font-medium transition-all"
   951→                style={{
   952→                  borderColor: "var(--border)",
   953→                  color: "var(--foreground)",
   954→                  background: "transparent",
   955→                }}
   956→              >
   957→                Download now
   958→              </button>
   959→            </div>
   960→          </div>
   961→
   962→          <div
   963→            className="flex items-start gap-2 rounded-lg border p-3 text-xs"
   964→            style={{
   965→              borderColor: "var(--border)",
   966→              background: "var(--muted)",
   967→              color: "var(--muted-foreground)",
   968→            }}
   969→          >
   970→            <Save className="size-4 shrink-0" style={{ color: "var(--cat-compress)" }} />
   971→            <p>
   972→              Pick a tool, then click/drag on the image. Annotations are rasterized onto the canvas

   974→              {imageFormat.toUpperCase()} via <code className="rounded px-1" style={{ background: "var(--card)" }}>canvas.toBlob</code>.
   975→            </p>
   976→          </div>
   977→        </div>
   978→      ) : target ? (

   980→          className="mt-5 rounded-xl border border-dashed p-12 text-center text-sm"

  1001→    </ToolPageShell>
