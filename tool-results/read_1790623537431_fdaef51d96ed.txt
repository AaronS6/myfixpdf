     6→/**
     7→ * Real client-side PDF compression pipeline:
     8→ *  - Re-loads the PDF with pdfjs.
     9→ *  - Rasterizes each page to a JPEG at a target DPI / quality.
    10→ *  - Reconstructs a new PDF with pdf-lib, embedding each JPEG as a page sized
    11→ *    to match the original page dimensions (in PDF points, 72 DPI).
    12→ *
    13→ * This produces genuinely smaller output for image-heavy or scanned PDFs, and
    14→ * applies true downsampling + JPEG re-encoding to all raster content. For
    15→ * pure-vector PDFs with no embedded images, the result is often similar to
    16→ * the original — which we surface to the user with a friendly note.
    17→ *
    18→ * Levels:
    19→ *   - "recommended":  ~110 DPI, JPEG quality 0.70  → balanced
    20→ *   - "extreme":      ~80 DPI,  JPEG quality 0.45  → smallest
    21→ *   - "light":        ~150 DPI, JPEG quality 0.92  → best quality
    22→ */
    23→
    24→export type CompressionLevel = "light" | "recommended" | "extreme";
    25→
    26→const LEVEL_PRESETS: Record<CompressionLevel, { dpi: number; quality: number; label: string; desc: string }> = {
    27→  light: { dpi: 150, quality: 0.92, label: "Best quality", desc: "Larger file, best quality" },
    28→  recommended: { dpi: 110, quality: 0.70, label: "Recommended", desc: "Balanced size & quality" },
    29→  extreme: { dpi: 80, quality: 0.45, label: "Smallest size", desc: "Smaller file, lower quality" },
    30→};
    31→
    32→export const COMPRESSION_LEVELS = LEVEL_PRESETS;
    33→
    34→export async function compressPdf(
    35→  blob: Blob,
    36→  level: CompressionLevel,
    37→  onProgress?: (pct: number, message: string) => void,
    38→): Promise<Uint8Array> {
    39→  const { loadPdfFromBlob } = await import("./pdfjs");
    40→  const preset = LEVEL_PRESETS[level];
    41→  const doc = await loadPdfFromBlob(blob);
    42→  const total = doc.numPages;
    43→  const images: Array<{ bytes: Uint8Array; width: number; height: number; pageW: number; pageH: number; isJpg: boolean }> = [];
    44→
    45→  for (let i = 1; i <= total; i++) {
    46→    const page = await doc.getPage(i);
    47→    const viewport = page.getViewport({ scale: 1 });
    48→    const pageW = viewport.width; // in PDF points (1pt = 1/72 inch)
    49→    const pageH = viewport.height;
    50→    // Convert target DPI to render scale: scale = targetDpi / 72
    51→    const scale = preset.dpi / 72;
    52→    const renderVp = page.getViewport({ scale });
    53→    const canvas = document.createElement("canvas");
    54→    canvas.width = renderVp.width;
    55→    canvas.height = renderVp.height;
    56→    const ctx = canvas.getContext("2d", { alpha: false });
    57→    if (!ctx) throw new Error("Canvas 2D context unavailable");
    58→    ctx.fillStyle = "#FFFFFF";
    59→    ctx.fillRect(0, 0, canvas.width, canvas.height);
    60→    // @ts-expect-error pdfjs legacy render context
    61→    await page.render({ canvasContext: ctx, viewport: renderVp }).promise;
    62→    const dataUrl = canvas.toDataURL("image/jpeg", preset.quality);
    63→    const bytes = dataUrlToBytes(dataUrl);
    64→    images.push({
    65→      bytes,
    66→      width: canvas.width,
    67→      height: canvas.height,
    68→      pageW,
    69→      pageH,
    70→      isJpg: true,
    71→    });
    72→    if (onProgress) {
    73→      onProgress(Math.round((i / total) * 70), `Rendering page ${i} of ${total}…`);
    74→    }
    75→  }
    76→  try {
    77→    // pdfjs v6 PDFDocumentProxy exposes cleanup(), not destroy()
    78→    await (doc as any).cleanup?.();
    79→  } catch {
    80→    // ignore
    81→  }
    82→
    83→  // Rebuild PDF with each JPEG embedded as a page sized to match the original page dimensions.
    84→  const out = await PDFDocument.create();
    85→  for (let i = 0; i < images.length; i++) {
    86→    const item = images[i];
    87→    const img = await out.embedJpg(item.bytes);
    88→    const page = out.addPage([item.pageW, item.pageH]);
    89→    page.drawImage(img, {
    90→      x: 0,
    91→      y: 0,
    92→      width: item.pageW,
    93→      height: item.pageH,
    94→    });
    95→    if (onProgress) {
    96→      const pct = 70 + Math.round(((i + 1) / images.length) * 25);
    97→      onProgress(pct, `Embedding compressed page ${i + 1}…`);
    98→    }
    99→  }
   100→  const bytes = await out.save({ useObjectStreams: true });
   101→  if (onProgress) onProgress(100, "Done");
   102→  return bytes;
   103→}
   104→
   105→function dataUrlToBytes(dataUrl: string): Uint8Array {
   106→  const base64 = dataUrl.split(",")[1] ?? "";
   107→  const bin = atob(base64);
   108→  const arr = new Uint8Array(bin.length);
   109→  for (let i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i);
   110→  return arr;
   111→}

   113→// ====== Split PDF ======
   114→export async function splitPdfSelectPages(
   115→  blob: Blob,
   116→  pageIndices: number[], // 0-based
   117→  onProgress?: (pct: number, message: string) => void,
   118→): Promise<Uint8Array> {
   119→  const src = await PDFDocument.load(await blob.arrayBuffer());
   120→  const out = await PDFDocument.create();
   121→  const copied = await out.copyPages(src, pageIndices);
   122→  for (const p of copied) out.addPage(p);
   123→  if (onProgress) onProgress(100, "Done");
   124→  return out.save({ useObjectStreams: true });
   125→}

   127→export async function splitPdfIntoGroups(
   128→  blob: Blob,
   129→  groups: number[][], // each group is an array of 0-based page indices
   130→  onProgress?: (pct: number, message: string) => void,
   131→): Promise<Uint8Array[]> {
   132→  const src = await PDFDocument.load(await blob.arrayBuffer());
   133→  const result: Uint8Array[] = [];
   134→  for (let g = 0; g < groups.length; g++) {
   135→    const out = await PDFDocument.create();
   136→    const copied = await out.copyPages(src, groups[g]);
   137→    for (const p of copied) out.addPage(p);
   138→    result.push(await out.save({ useObjectStreams: true }));
   139→    if (onProgress) onProgress(Math.round(((g + 1) / groups.length) * 100), `Building group ${g + 1}…`);
   140→  }
   141→  return result;
   142→}

   144→// ====== Rotate a single page (does NOT rotate all pages — only the requested index) ======
   145→export async function rotatePdfPage(
   146→  blob: Blob,
   147→  pageIndex: number, // 0-based
   148→  rotation: 90 | 180 | 270,
   149→  onProgress?: (pct: number, message: string) => void,
   150→): Promise<Uint8Array> {
   151→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   152→  const pages = doc.getPages();
   153→  if (pageIndex < 0 || pageIndex >= pages.length) {
   154→    throw new Error(`Page ${pageIndex + 1} does not exist in this PDF.`);
   155→  }
   156→  const page = pages[pageIndex];
   157→  const current = page.getRotation().angle;
   158→  page.setRotation(degrees((current + rotation) % 360));
   159→  if (onProgress) onProgress(100, "Done");
   160→  return doc.save({ useObjectStreams: true });
   161→}

   218→// ====== Add text to a page at given coords ======
   219→export async function addTextToPage(
   220→  blob: Blob,
   221→  pageIndex: number,
   222→  text: string,
   223→  opts: { x: number; y: number; size: number; color: [number, number, number]; font?: "Helvetica" | "TimesRoman" },
   224→  onProgress?: (pct: number, message: string) => void,

   233→  page.drawText(text, {
   234→    x: opts.x,
   235→    y: opts.y,
   236→    size: opts.size,
   237→    font,
   238→    color: rgb(opts.color[0], opts.color[1], opts.color[2]),
   239→  });
   240→  if (onProgress) onProgress(100, "Done");
   241→  return doc.save({ useObjectStreams: true });

   262→/**
   263→ * Embed a rasterized stroke PNG onto a PDF page at the stroke's ACTUAL
   264→ * bounding-box position (not stretched to full A4). Handles the Y-axis
   265→ * flip: stroke points are in screen coords (top-left origin, Y down),
   266→ * but pdf-lib's drawImage uses PDF coords (bottom-left origin, Y up).
   267→ *
   268→ * `strokePng` carries the PNG bytes plus the bounding box in PDF points
   269→ * (minX/minY/maxX/maxY), the canvas dimensions (w/h), and the padding
   270→ * that was added around the stroke when rasterizing.
   271→ */
   272→export async function embedStrokeOnPage(
   273→  blob: Blob,
   274→  pageIndex: number,
   275→  strokePng: { bytes: Uint8Array; minX: number; minY: number; maxX: number; maxY: number; w: number; h: number; pad: number },
   276→  onProgress?: (pct: number, message: string) => void,
   277→): Promise<Uint8Array> {
   278→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   279→  const page = doc.getPages()[pageIndex];
   280→  if (!page) throw new Error(`Page ${pageIndex + 1} does not exist.`);
   281→  const pageHeight = page.getHeight();
   282→  if (strokePng.bytes.length === 0) return doc.save({ useObjectStreams: true });
   283→  const img = await doc.embedPng(strokePng.bytes);
   284→  // The PNG covers the stroke's bounding box in screen coords:
   285→  //   x ∈ [minX - pad, maxX + pad],  y ∈ [minY - pad, maxY + pad]
   286→  // In PDF coords (Y up from bottom), the image's bottom-left corner is at:
   287→  //   x = minX - pad
   288→  //   y = pageHeight - (maxY + pad)   (flip: top of stroke → measured from bottom)
   289→  page.drawImage(img, {
   290→    x: strokePng.minX - strokePng.pad,
   291→    y: pageHeight - (strokePng.maxY + strokePng.pad),
   292→    width: strokePng.w,
   293→    height: strokePng.h,
   294→  });
   295→  if (onProgress) onProgress(100, "Done");
   296→  return doc.save({ useObjectStreams: true });
   297→}

   314→// ====== JPG/PNG → single PDF ======
   315→export async function imagesToPdf(
   316→  images: Array<{ bytes: Uint8Array; format: "png" | "jpg"; width: number; height: number }>,
   317→  opts: { pageSize: "fit" | "a4" | "letter"; orientation: "portrait" | "landscape"; margin: number },
   318→  onProgress?: (pct: number, message: string) => void,
   319→): Promise<Uint8Array> {
   320→  const doc = await PDFDocument.create();
   321→  const A4 = [595.28, 841.89];
   322→  const Letter = [612, 792];
   323→  const base = opts.pageSize === "a4" ? A4 : opts.pageSize === "letter" ? Letter : null;
   324→  for (let i = 0; i < images.length; i++) {
   325→    const item = images[i];
   326→    const img = item.format === "png" ? await doc.embedPng(item.bytes) : await doc.embedJpg(item.bytes);
   327→    let pageW: number, pageH: number;
   328→    if (base) {
   329→      if (opts.orientation === "landscape") {
   330→        pageW = base[1];
   331→        pageH = base[0];
   332→      } else {
   333→        pageW = base[0];
   334→        pageH = base[1];
   335→      }
   336→    } else {
   337→      // Fit to image
   338→      pageW = item.width;
   339→      pageH = item.height;
   340→    }
   341→    const page = doc.addPage([pageW, pageH]);
   342→    const margin = opts.margin;
   343→    const availW = pageW - margin * 2;
   344→    const availH = pageH - margin * 2;
   345→    const scale = Math.min(availW / item.width, availH / item.height);
   346→    const drawW = item.width * scale;
   347→    const drawH = item.height * scale;
   348→    const x = (pageW - drawW) / 2;
   349→    const y = (pageH - drawH) / 2;
   350→    page.drawImage(img, { x, y, width: drawW, height: drawH });
   351→    if (onProgress) onProgress(Math.round(((i + 1) / images.length) * 100), `Embedding image ${i + 1}…`);
   352→  }
   353→  return doc.save({ useObjectStreams: true });
   354→}

   356→// ====== Merge mixed PDF + images into one PDF ======
   357→export type MergeItem =
   358→  | { kind: "pdf-page"; blob: Blob; pageIndex: number }
   359→  | { kind: "image"; bytes: Uint8Array; format: "png" | "jpg"; width: number; height: number };
   360→
   361→export async function mergeItemsToPdf(
   362→  items: MergeItem[],
   363→  onProgress?: (pct: number, message: string) => void,
   364→): Promise<Uint8Array> {
   365→  const out = await PDFDocument.create();
   366→  const pdfCache = new Map<Blob, PDFDocument>();
   367→  const MAX_IMG_DIM = 2000; // downscale large images to keep the PDF small + fast
   368→
   369→  for (let i = 0; i < items.length; i++) {
   370→    const item = items[i];
   371→
   372→    if (item.kind === "pdf-page") {
   373→      let src = pdfCache.get(item.blob);
   374→      if (!src) {
   375→        src = await PDFDocument.load(await item.blob.arrayBuffer());
   376→        pdfCache.set(item.blob, src);
   377→      }
   378→      const [copied] = await out.copyPages(src, [item.pageIndex]);
   379→      out.addPage(copied);
   380→    } else {
   381→      // For images: if the image is very large, downscale it via canvas first
   382→      // to keep the PDF small and the embedding fast. pdf-lib's embedPng/embedJpg
   383→      // has to process the full image data — a 4000×3000 PNG is ~48MB of raw
   384→      // pixel data that slows everything down.
   385→      let embedBytes = item.bytes;
   386→      let embedW = item.width;
   387→      let embedH = item.height;
   388→      const longest = Math.max(item.width, item.height);
   389→      if (longest > MAX_IMG_DIM) {
   390→        const scale = MAX_IMG_DIM / longest;
   391→        embedW = Math.round(item.width * scale);
   392→        embedH = Math.round(item.height * scale);
   393→        // Downscale via canvas
   394→        const blob = new Blob([item.bytes as unknown as BlobPart], { type: item.format === "png" ? "image/png" : "image/jpeg" });
   395→        const url = URL.createObjectURL(blob);
   396→        try {
   397→          const img = await new Promise<HTMLImageElement>((resolve, reject) => {
   398→            const im = new Image();
   399→            im.onload = () => resolve(im);
   400→            im.onerror = () => reject(new Error("Failed to load image for downscaling"));
   401→            im.src = url;
   402→          });
   403→          const canvas = document.createElement("canvas");
   404→          canvas.width = embedW;
   405→          canvas.height = embedH;
   406→          const ctx = canvas.getContext("2d");
   407→          if (ctx) {
   408→            ctx.drawImage(img, 0, 0, embedW, embedH);
   409→            const reencoded: Blob = await new Promise((resolve, reject) =>
   410→              canvas.toBlob(
   411→                (b) => (b ? resolve(b) : reject(new Error("Re-encode failed"))),
   412→                item.format === "png" ? "image/png" : "image/jpeg",
   413→                0.9,
   414→              ),
   415→            );
   416→            embedBytes = new Uint8Array(await reencoded.arrayBuffer());
   417→          }
   418→        } finally {
   419→          URL.revokeObjectURL(url);
   420→        }
   421→      }
   422→
   423→      const img = item.format === "png"
   424→        ? await out.embedPng(embedBytes)
   425→        : await out.embedJpg(embedBytes);
   426→      const page = out.addPage([embedW, embedH]);
   427→      page.drawImage(img, { x: 0, y: 0, width: embedW, height: embedH });
   428→    }
   429→
   430→    if (onProgress) onProgress(Math.round(((i + 1) / items.length) * 100), `Merging ${i + 1} of ${items.length}…`);
   431→
   432→    // Yield to the event loop every 3 items so the UI doesn't freeze.
   433→    // This lets the ProgressOverlay update and the browser stay responsive.
   434→    if (i % 3 === 2) {
   435→      await new Promise((r) => setTimeout(r, 0));
   436→    }
   437→  }
   438→
   439→  return out.save({ useObjectStreams: true });
   440→}

   442→// ====== PDF → JPG images (one per page) ======
   443→export async function pdfToJpgImages(
   444→  blob: Blob,
   445→  scale = 1.5,
   446→  quality = 0.85,
   447→  onProgress?: (pct: number, message: string) => void,
   448→): Promise<Array<{ bytes: Uint8Array; name: string; width: number; height: number; previewUrl: string }>> {
   449→  const { loadPdfFromBlob } = await import("./pdfjs");
   450→  const doc = await loadPdfFromBlob(blob);
   451→  const total = doc.numPages;
   452→  const out: Array<{ bytes: Uint8Array; name: string; width: number; height: number; previewUrl: string }> = [];
   453→  for (let i = 1; i <= total; i++) {
   454→    const page = await doc.getPage(i);
   455→    const viewport = page.getViewport({ scale });
   456→    const canvas = document.createElement("canvas");
   457→    canvas.width = viewport.width;
   458→    canvas.height = viewport.height;
   459→    const ctx = canvas.getContext("2d", { alpha: false });
   460→    if (!ctx) throw new Error("Canvas 2D context unavailable");
   461→    ctx.fillStyle = "#FFFFFF";
   462→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   463→    // @ts-expect-error pdfjs legacy render context
   464→    await page.render({ canvasContext: ctx, viewport }).promise;
   465→    const blob2: Blob = await new Promise((resolve) =>
   466→      canvas.toBlob((b) => resolve(b as Blob), "image/jpeg", quality),
   467→    );
   468→    const url = URL.createObjectURL(blob2);
   469→    out.push({
   470→      bytes: new Uint8Array(await blob2.arrayBuffer()),
   471→      name: `page-${i}.jpg`,
   472→      width: canvas.width,
   473→      height: canvas.height,
   474→      previewUrl: url,
   475→    });
   476→    if (onProgress) onProgress(Math.round((i / total) * 100), `Rendering page ${i}…`);
   477→  }
   478→  try {
   479→    await (doc as any).cleanup?.();
   480→  } catch {
   481→    // ignore
   482→  }
   483→  return out;
   484→}

   486→// ====== PDF → PNG images (one per page) ======
   487→export async function pdfToPngImages(
   488→  blob: Blob,
   489→  scale = 1.5,
   490→  onProgress?: (pct: number, message: string) => void,
   491→): Promise<Array<{ bytes: Uint8Array; name: string; width: number; height: number; previewUrl: string }>> {
   492→  const { loadPdfFromBlob } = await import("./pdfjs");
   493→  const doc = await loadPdfFromBlob(blob);
   494→  const total = doc.numPages;
   495→  const out: Array<{ bytes: Uint8Array; name: string; width: number; height: number; previewUrl: string }> = [];
   496→  for (let i = 1; i <= total; i++) {
   497→    const page = await doc.getPage(i);
   498→    const viewport = page.getViewport({ scale });
   499→    const canvas = document.createElement("canvas");
   500→    canvas.width = viewport.width;
   501→    canvas.height = viewport.height;
   502→    const ctx = canvas.getContext("2d");
   503→    if (!ctx) throw new Error("Canvas 2D context unavailable");
   504→    // White background so transparent PDF regions don't end up as alpha-only
   505→    ctx.fillStyle = "#FFFFFF";
   506→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   507→    // @ts-expect-error pdfjs legacy render context
   508→    await page.render({ canvasContext: ctx, viewport }).promise;
   509→    const blob2: Blob = await new Promise((resolve, reject) => {
   510→      canvas.toBlob(
   511→        (b) => (b ? resolve(b) : reject(new Error("PNG encoding failed"))),
   512→        "image/png",
   513→      );
   514→    });
   515→    const url = URL.createObjectURL(blob2);
   516→    out.push({
   517→      bytes: new Uint8Array(await blob2.arrayBuffer()),
   518→      name: `page-${i}.png`,
   519→      width: canvas.width,
   520→      height: canvas.height,
   521→      previewUrl: url,
   522→    });
   523→    if (onProgress) onProgress(Math.round((i / total) * 100), `Rendering page ${i}…`);
   524→  }
   525→  try {
   526→    await (doc as any).cleanup?.();
   527→  } catch {
   528→    // ignore
   529→  }
   530→  return out;
   531→}

   533→// ====== Add a text watermark to every page ======
   534→export type WatermarkOptions = {
   535→  text: string;
   536→  fontSize: number; // pt
   537→  opacity: number; // 0..1
   538→  rotation: number; // degrees
   539→  color: [number, number, number]; // 0..1 each
   540→  /** "all" | "first" | "last" | number (specific 0-based page) */
   541→  target: "all" | "first" | "last" | number;
   542→  /** "center" | "tile" (grid) | corner positions */
   543→  position: "center" | "tile" | "top-left" | "top-right" | "bottom-left" | "bottom-right";
   544→};
   545→
   546→export async function addTextWatermark(
   547→  blob: Blob,
   548→  opts: WatermarkOptions,
   549→  onProgress?: (pct: number, message: string) => void,
   550→): Promise<Uint8Array> {
   551→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   552→  const font = await doc.embedFont(StandardFonts.HelveticaBold);
   553→  const pages = doc.getPages();
   554→  const targetIndices = computeTargetIndices(opts.target, pages.length);
   555→  for (let i = 0; i < pages.length; i++) {
   556→    if (!targetIndices.has(i)) continue;
   557→    const page = pages[i];
   558→    const { width, height } = page.getSize();
   559→    const positions = computeWatermarkPositions(opts.position, width, height, opts.fontSize, opts.text.length);
   560→    for (const pos of positions) {
   561→      page.drawText(opts.text, {
   562→        x: pos.x,
   563→        y: pos.y,
   564→        size: opts.fontSize,
   565→        font,
   566→        color: rgb(opts.color[0], opts.color[1], opts.color[2]),
   567→        opacity: opts.opacity,
   568→        rotate: degrees(opts.rotation),
   569→      });
   570→    }
   571→    if (onProgress) onProgress(Math.round(((i + 1) / pages.length) * 100), `Watermarking page ${i + 1}…`);
   572→  }
   573→  return doc.save({ useObjectStreams: true });
   574→}
   575→
   576→function computeTargetIndices(target: WatermarkOptions["target"], total: number): Set<number> {
   577→  if (target === "all") return new Set(Array.from({ length: total }, (_, i) => i));
   578→  if (target === "first") return new Set([0]);
   579→  if (target === "last") return new Set([total - 1]);
   580→  if (typeof target === "number") return new Set([Math.max(0, Math.min(total - 1, target))]);
   581→  return new Set();
   582→}
   583→
   584→function computeWatermarkPositions(
   585→  position: WatermarkOptions["position"],
   586→  pageW: number,
   587→  pageH: number,
   588→  fontSize: number,
   589→  textLen: number,
   590→): Array<{ x: number; y: number }> {
   591→  const textW = textLen * fontSize * 0.55; // approximate
   592→  const textH = fontSize;
   593→  if (position === "center") {
   594→    return [{ x: (pageW - textW) / 2, y: (pageH - textH) / 2 }];
   595→  }
   596→  if (position === "top-left") return [{ x: 40, y: pageH - textH - 40 }];
   597→  if (position === "top-right") return [{ x: pageW - textW - 40, y: pageH - textH - 40 }];
   598→  if (position === "bottom-left") return [{ x: 40, y: 40 }];
   599→  if (position === "bottom-right") return [{ x: pageW - textW - 40, y: 40 }];
   600→  // tile — 3×3 grid
   601→  const out: Array<{ x: number; y: number }> = [];
   602→  for (let row = 0; row < 3; row++) {
   603→    for (let col = 0; col < 3; col++) {
   604→      const x = (pageW / 3) * col + (pageW / 6) - textW / 2;
   605→      const y = (pageH / 3) * row + (pageH / 6) - textH / 2;
   606→      out.push({ x, y });
   607→    }
   608→  }
   609→  return out;
   610→}

   612→// ====== Add image watermark (logo/stamp) ======
   613→export async function addImageWatermark(
   614→  blob: Blob,
   615→  imageBytes: Uint8Array,
   616→  imageFormat: "png" | "jpg",
   617→  opts: { opacity: number; scale: number; position: WatermarkOptions["position"]; target: WatermarkOptions["target"] },
   618→  onProgress?: (pct: number, message: string) => void,
   619→): Promise<Uint8Array> {
   620→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   621→  const img = imageFormat === "png" ? await doc.embedPng(imageBytes) : await doc.embedJpg(imageBytes);
   622→  const pages = doc.getPages();
   623→  const targetIndices = computeTargetIndices(opts.target, pages.length);
   624→  const imgW = img.width * opts.scale;
   625→  const imgH = img.height * opts.scale;
   626→  for (let i = 0; i < pages.length; i++) {
   627→    if (!targetIndices.has(i)) continue;
   628→    const page = pages[i];
   629→    const { width, height } = page.getSize();
   630→    let x = (width - imgW) / 2;
   631→    let y = (height - imgH) / 2;
   632→    if (opts.position === "top-left") { x = 40; y = height - imgH - 40; }
   633→    else if (opts.position === "top-right") { x = width - imgW - 40; y = height - imgH - 40; }
   634→    else if (opts.position === "bottom-left") { x = 40; y = 40; }
   635→    else if (opts.position === "bottom-right") { x = width - imgW - 40; y = 40; }
   636→    page.drawImage(img, { x, y, width: imgW, height: imgH, opacity: opts.opacity });
   637→    if (onProgress) onProgress(Math.round(((i + 1) / pages.length) * 100), `Stamping page ${i + 1}…`);
   638→  }
   639→  return doc.save({ useObjectStreams: true });
   640→}

   642→// ====== Crop a single PDF page (trim margins to a specified rectangle) ======
   643→export async function cropPdfPage(
   644→  blob: Blob,
   645→  pageIndex: number,
   646→  crop: { left: number; right: number; top: number; bottom: number }, // each in PDF points
   647→  onProgress?: (pct: number, message: string) => void,
   648→): Promise<Uint8Array> {
   649→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   650→  const page = doc.getPages()[pageIndex];
   651→  if (!page) throw new Error(`Page ${pageIndex + 1} does not exist.`);
   652→  const { width, height } = page.getSize();
   653→  const newW = Math.max(40, width - crop.left - crop.right);
   654→  const newH = Math.max(40, height - crop.top - crop.bottom);
   655→  page.setCropBox(crop.left, crop.bottom, newW, newH);
   656→  page.setMediaBox(crop.left, crop.bottom, newW, newH);
   657→  if (onProgress) onProgress(100, "Done");
   658→  return doc.save({ useObjectStreams: true });
   659→}

   661→// ====== Add page numbers to every page (footer center) ======
   662→export async function addPageNumbers(
   663→  blob: Blob,
   664→  opts: { format: "1/3" | "Page 1 of 3" | "1" | "1 of 3"; fontSize: number; position: "bottom-center" | "bottom-right" | "top-center" | "top-right"; color: [number, number, number]; startFrom: number },
   665→  onProgress?: (pct: number, message: string) => void,
   666→): Promise<Uint8Array> {
   667→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   668→  const font = await doc.embedFont(StandardFonts.Helvetica);
   669→  const pages = doc.getPages();
   670→  const total = pages.length;
   671→  for (let i = 0; i < total; i++) {
   672→    const page = pages[i];
   673→    const num = i + opts.startFrom;
   674→    const text = opts.format
   675→      .replace("1", String(num))
   676→      .replace("3", String(total + opts.startFrom - 1));
   677→    const textW = font.widthOfTextAtSize(text, opts.fontSize);
   678→    const { width, height } = page.getSize();
   679→    let x = (width - textW) / 2;
   680→    let y = 20;
   681→    if (opts.position === "bottom-right") { x = width - textW - 30; y = 20; }
   682→    else if (opts.position === "top-center") { x = (width - textW) / 2; y = height - opts.fontSize - 20; }
   683→    else if (opts.position === "top-right") { x = width - textW - 30; y = height - opts.fontSize - 20; }
   684→    page.drawText(text, {
   685→      x, y, size: opts.fontSize, font,
   686→      color: rgb(opts.color[0], opts.color[1], opts.color[2]),
   687→    });
   688→    if (onProgress) onProgress(Math.round(((i + 1) / total) * 100), `Numbering page ${i + 1}…`);
   689→  }
   690→  return doc.save({ useObjectStreams: true });
   691→}

   693→// ====== Rotate ALL pages in one pass (true bulk rotation) ======
   694→export async function rotateAllPages(
   695→  blob: Blob,
   696→  rotation: 90 | 180 | 270,
   697→  onProgress?: (pct: number, message: string) => void,
   698→): Promise<Uint8Array> {
   699→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   700→  const pages = doc.getPages();
   701→  const total = pages.length;
   702→  for (let i = 0; i < total; i++) {
   703→    const page = pages[i];
   704→    const current = page.getRotation().angle;
   705→    // Add rotation modulo 360 so we never exceed 360.
   706→    page.setRotation(degrees((current + rotation) % 360));
   707→    if (onProgress && total > 0) {
   708→      onProgress(Math.round(((i + 1) / total) * 100), `Rotating page ${i + 1} of ${total}…`);
   709→    }
   710→  }
   711→  return doc.save({ useObjectStreams: true });
   712→}

   714→// ====== Redact pages — bake filled-black rectangles over sensitive text ======
   715→// The redaction is TRUE: the underlying text/vector content remains in the
   716→// PDF but is visually covered by an opaque black rectangle rendered ABOVE it.
   717→// Coordinates are in PDF points; y is measured from the TOP-left of the page
   718→// (matches the on-screen canvas drag convention). We convert to pdf-lib's
   719→// bottom-left origin internally.
   720→export type Redaction = {
   721→  pageIndex: number; // 0-based
   722→  x: number; // top-left X in PDF points
   723→  y: number; // top-left Y from the top of the page
   724→  width: number;
   725→  height: number;
   726→};
   727→
   728→export async function redactPdfPages(
   729→  blob: Blob,
   730→  redactions: Redaction[],
   731→  onProgress?: (pct: number, message: string) => void,
   732→): Promise<Uint8Array> {
   733→  const doc = await PDFDocument.load(await blob.arrayBuffer());
   734→  const pages = doc.getPages();
   735→  // Group by page so we only resolve page size once per page.
   736→  const byPage = new Map<number, Redaction[]>();
   737→  for (const r of redactions) {
   738→    if (r.pageIndex < 0 || r.pageIndex >= pages.length) continue;
   739→    if (!byPage.has(r.pageIndex)) byPage.set(r.pageIndex, []);
   740→    byPage.get(r.pageIndex)!.push(r);
   741→  }
   742→  const total = redactions.length || 1;
   743→  let done = 0;
   744→  for (const [pageIndex, rects] of byPage) {
   745→    const page = pages[pageIndex];
   746→    const { height } = page.getSize();
   747→    for (const r of rects) {
   748→      // Convert top-left Y to bottom-left Y for pdf-lib.
   749→      const yFromBottom = height - r.y - r.height;
   750→      page.drawRectangle({
   751→        x: r.x,
   752→        y: yFromBottom,
   753→        width: r.width,
   754→        height: r.height,
   755→        color: rgb(0, 0, 0),
   756→      });
   757→      done++;
   758→      if (onProgress) {
   759→        onProgress(Math.round((done / total) * 100), `Baking redaction ${done} of ${total}…`);
   760→      }
   761→    }
   762→  }
   763→  return doc.save({ useObjectStreams: true });
   764→}

   766→// ====== Rebuild a brand-new PDF from translated text (one page → one or more pages) ======
   767→// The original layout is NOT preserved — we word-wrap the translated text into
   768→// A4 pages with the Helvetica font. This is approximate by design.
   769→export async function rebuildPdfWithText(
   770→  pages: Array<{ text: string }>,
   771→  opts: { fontSize?: number; margin?: number; lineHeight?: number } = {},
   772→  onProgress?: (pct: number, message: string) => void,
   773→): Promise<Uint8Array> {
   774→  const doc = await PDFDocument.create();
   775→  const font = await doc.embedFont(StandardFonts.Helvetica);
   776→  const fontSize = opts.fontSize ?? 12;
   777→  const margin = opts.margin ?? 40;
   778→  const lineH = opts.lineHeight ?? fontSize * 1.4;
   779→  // A4 portrait in PDF points (72 DPI).
   780→  const pageW = 595.28;
   781→  const pageH = 841.89;
   782→  const maxW = pageW - margin * 2;
   783→
   784→  // Wrap text into display lines, preserving blank-line paragraph breaks.
   785→  const wrapParagraph = (paragraph: string): string[] => {
   786→    const words = paragraph.split(/\s+/).filter((w) => w.length > 0);
   787→    if (words.length === 0) return [""];
   788→    const out: string[] = [];
   789→    let line = "";
   790→    for (const w of words) {
   791→      const test = line ? `${line} ${w}` : w;
   792→      if (font.widthOfTextAtSize(test, fontSize) > maxW) {
   793→        if (line) out.push(line);
   794→        line = w;
   795→      } else {
   796→        line = test;
   797→      }
   798→    }
   799→    if (line) out.push(line);
   800→    return out;
   801→  };
   802→
   803→  for (let i = 0; i < pages.length; i++) {
   804→    let page = doc.addPage([pageW, pageH]);
   805→    let y = pageH - margin - fontSize;
   806→    const text = pages[i].text || "";
   807→    // Preserve paragraph breaks (\n) as blank lines.
   808→    const paragraphs = text.split(/\n/);
   809→    const allLines: string[] = [];
   810→    for (const para of paragraphs) {
   811→      const lines = wrapParagraph(para);
   812→      for (const ln of lines) allLines.push(ln);
   813→    }
   814→
   815→    for (const ln of allLines) {
   816→      if (y < margin) {
   817→        page = doc.addPage([pageW, pageH]);
   818→        y = pageH - margin - fontSize;
   819→      }
   820→      if (ln) {
   821→        try {
   822→          page.drawText(ln, {
   823→            x: margin,
   824→            y,
   825→            size: fontSize,
   826→            font,
   827→            color: rgb(0, 0, 0),
   828→          });
   829→        } catch {
   830→          // Skip any un-encodable characters (Helvetica StandardFont is WinAnsi).
   831→        }
   832→      }
   833→      y -= lineH;
   834→    }
   835→
   836→    if (onProgress) {
   837→      onProgress(Math.round(((i + 1) / pages.length) * 100), `Building PDF page ${i + 1} of ${pages.length}…`);
   838→    }
   839→  }
   840→  return doc.save({ useObjectStreams: true });
   841→}
